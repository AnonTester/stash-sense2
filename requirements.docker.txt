# Docker runtime requirements

# Face recognition
# onnxruntime itself is intentionally NOT pinned here -- each Dockerfile
# variant (CPU/Dockerfile, Dockerfile.rocm, Dockerfile.cuda) installs its
# own vendor-specific build as a separate, ordered step after this file,
# since insightface's own declared onnxruntime dependency would otherwise
# silently clobber a GPU-specific build installed first. See Dockerfile.rocm's
# comments for the full story (proven live in /opt/face-pipeline-bench).
insightface>=0.7.3,<1
onnx>=1.15.0,<1.18  # insightface pulls this in unpinned; onnx>=1.18 registers a new
                     # ml_dtypes dtype at import whose finfo() segfaults under numpy<2
ml_dtypes>=0.4.0,<0.5
usearch>=2.16.0,<2.25  # 2.25+ switched its SIMD backend to numkong, whose mere import
                        # destabilizes process state and segfaults onnxruntime later
                        # when it creates a CPU inference session

# API
fastapi>=0.100.0,<1
uvicorn[standard]>=0.23.0,<1
httpx>=0.25.0,<1

opencv-python-headless>=4.8.0,<5
